    function shareResult() {
      const item = TYPES_32[state.resultCode];
      const pageUrl = getShareableUrl(item.code);
      const shareText = `ゴンチャ32タイプ診断の結果は【${item.code}：${item.persona}】でした！\nおすすめの一杯：${item.drinkName}\n注文カスタム：${item.custom}\n#ゴンチャ32タイプ診断 #Gongcha\n${pageUrl}`;
      const shareUrl = `https://twitter.com/intent/tweet?text=${encodeURIComponent(shareText)}`;
      window.open(shareUrl, '_blank', 'noopener,noreferrer');
    }

    function getShareableUrl(code) {
      const base = window.location.href.split('?')[0].split('#')[0];
      if (base.startsWith('http://') || base.startsWith('https://')) {
        return `${base}?type=${code}`;
      }
      return '';
    }

    function copyShareUrl() {
      const url = getShareableUrl(state.resultCode);
      if (!url) {
        showToast('公開URL発行後に結果リンクをコピーできます');
        return;
      }
      copyCustomText(url);
      showToast('この診断結果のリンクをコピーしました！');
    }
